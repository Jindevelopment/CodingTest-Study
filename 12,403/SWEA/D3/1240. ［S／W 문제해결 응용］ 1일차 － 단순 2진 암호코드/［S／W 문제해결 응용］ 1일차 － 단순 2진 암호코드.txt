T = int(input())
my_dict = {"0001101":0,"0011001":1,"0010011":2,"0111101":3,"0100011":4,"0110001":5,"0101111":6,"0111011":7,
           "0110111":8,"0001011":9}

for ts in range(1,T+1):
    a, b = map(int, input().split())
    li = []
    for i in range(a):
        row = list(input())
        temp = []
        for k in row:
            temp.append(k)
        li.append(temp)
    
    last_x = 0
    last_y = 0
    found = False
    for x in range(a-1,-1,-1):
        for y in range(b-1,-1,-1):
            if li[x][y] == "1":
                last_x = x
                last_y = y
                found = True
                break
        if found==True:
            break
                
    total_even = 0
    total_odd = 0
    total = 0
    for u in range(last_y-55,last_y-6,14):
        total_even += my_dict["".join(li[last_x][u:u+7])]
    
    for v in range(last_y-48,last_y+1,14):
        total_odd += my_dict["".join(li[last_x][v:v+7])]
    
    total = total_odd  + total_even * 3
    if total%10==0:
        res = total_odd+total_even
    else:
        res = 0
    
    print(f"#{ts} {res}")
